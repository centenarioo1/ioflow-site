generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
}

enum Role {
  USER
  ADMIN
}

model User {
  id           String         @id @default(uuid())
  nome         String
  email        String         @unique
  password     String
  role         Role           @default(USER)
  pontos       Int            @default(0)
  createdAt    DateTime       @default(now())
  updatedAt    DateTime       @updatedAt
  orders       Order[]
  topicos      Topico[]
  respostas    Resposta[]
  mensagens    ChatMensagem[]
  presencaChat PresencaChat?
  cuponsUsados CupomUsado[]

  @@map("users")
}

model Order {
  id                  String       @id @default(cuid())
  userId              String?
  customerName        String
  customerEmail       String
  customerPhone       String
  customerCpf         String
  
  // Endereço de Entrega
  shippingCep         String
  shippingStreet      String
  shippingNumber      String
  shippingComplement  String?
  shippingNeighborhood String
  shippingCity        String
  shippingState       String
  shippingServiceName String?
  shippingCost        Float        @default(0)
  trackingCode        String?

  // Valores e Itens
  items               Json         // Array com produtos, variações, preço e qtd
  subtotal            Float
  desconto            Float        @default(0)
  total               Float
  cupomId             String?
  cupomCodigo         String?

  // Mercado Pago & Status
  status              String       @default("Pendente") // Pendente, Aprovado, Em Preparação, Enviado, Entregue, Cancelado
  mpPreferenceId      String?
  mpPaymentId         String?
  paymentMethod       String?

  createdAt           DateTime     @default(now())
  updatedAt           DateTime     @updatedAt

  cupom               Cupom?       @relation(fields: [cupomId], references: [id])
  user                User?        @relation(fields: [userId], references: [id])

  @@map("orders")
}

model Cupom {
  id            String       @id @default(cuid())
  codigo        String       @unique
  descricao     String
  tipo          String
  valor         Float
  valorMinimo   Float?
  usoMaximo     Int          @default(1)
  usoTotal      Int          @default(0)
  limiteTotal   Int?
  ativo         Boolean      @default(true)
  dataInicio    DateTime     @default(now())
  dataExpiracao DateTime?
  createdAt     DateTime     @default(now())
  orders        Order[]
  cuponsUsados  CupomUsado[]

  @@map("cupons")
}

model CupomUsado {
  id      String   @id @default(cuid())
  userId  String
  cupomId String
  usadoEm DateTime @default(now())
  cupom   Cupom    @relation(fields: [cupomId], references: [id], onDelete: Cascade)
  user    User     @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@map("cupons_usados")
}

model Topico {
  id           String     @id @default(cuid())
  titulo       String
  conteudo     String
  categoria    String
  imagens      String[]   @default([])
  autorId      String
  autor        User       @relation(fields: [autorId], references: [id], onDelete: Cascade)
  respostas    Resposta[]
  criadoEm     DateTime   @default(now())
  atualizadoEm DateTime   @updatedAt

  @@map("topicos")
}

model Resposta {
  id       String   @id @default(cuid())
  conteudo String
  imagens  String[] @default([])
  topicoId String
  topico   Topico   @relation(fields: [topicoId], references: [id], onDelete: Cascade)
  autorId  String
  autor    User     @relation(fields: [autorId], references: [id], onDelete: Cascade)
  criadoEm DateTime @default(now())

  @@map("respostas")
}

model ChatMensagem {
  id       String   @id @default(cuid())
  conteudo String
  autorId  String
  autor    User     @relation(fields: [autorId], references: [id], onDelete: Cascade)
  criadoEm DateTime @default(now())

  @@map("chat_mensagens")
}

model PresencaChat {
  id        String   @id @default(cuid())
  usuarioId String   @unique
  usuario   User     @relation(fields: [usuarioId], references: [id], onDelete: Cascade)
  vistoEm   DateTime @default(now())

  @@map("presenca_chat")
}
