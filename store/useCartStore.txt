// store/useCartStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

export interface CartItem {
  id: string;             // id único (produto + variação)
  productId?: string;
  variantId?: string;
  nome: string;           // nome do produto
  preco: number;          // preço unitário
  originalPrice?: number;
  variacao?: string;      // tamanho / peso escolhido (ex: "1kg", "500ml")
  imagem?: string;
  weight?: number;        // peso em gramas (para cálculo de frete)
  quantity: number;
}

// Formato flexível: aceita tanto os campos em PT (carrinho)
// quanto em EN (home/página de produto), normalizando internamente.
type AddInput = {
  id: string;
  productId?: string;
  variantId?: string;
  nome?: string;
  name?: string;
  preco?: number;
  price?: number;
  originalPrice?: number;
  variacao?: string;
  size?: string;
  imagem?: string;
  image?: string;
  weight?: number;
  quantity?: number;
};

function normalize(input: AddInput): Omit<CartItem, 'quantity'> {
  return {
    id: input.id,
    productId: input.productId,
    variantId: input.variantId,
    nome: input.nome ?? input.name ?? 'Produto',
    preco: Number(input.preco ?? input.price ?? 0),
    originalPrice: input.originalPrice,
    variacao: input.variacao ?? input.size,
    imagem: input.imagem ?? input.image,
    weight: input.weight,
  };
}

interface CartState {
  items: CartItem[];
  addToCart: (item: AddInput) => void;
  addItem: (item: AddInput, quantity?: number) => void; // alias (compatibilidade)
  removeItem: (id: string, variacao?: string) => void;
  updateQuantity: (id: string, quantity: number, variacao?: string) => void;
  clearCart: () => void;
  totalItems: () => number;
}

export const useCartStore = create<CartState>()(
  persist(
    (set, get) => ({
      items: [],

      addToCart: (input) =>
        set((state) => {
          const item = normalize(input);
          const qty = Number(input.quantity) || 1;
          const existente = state.items.find(
            (i) => i.id === item.id && i.variacao === item.variacao
          );
          if (existente) {
            return {
              items: state.items.map((i) =>
                i.id === item.id && i.variacao === item.variacao
                  ? { ...i, quantity: i.quantity + qty }
                  : i
              ),
            };
          }
          return { items: [...state.items, { ...item, quantity: qty }] };
        }),

      // Alias: qualquer código antigo que use addItem continua funcionando
      addItem: (input, quantity = 1) => {
        get().addToCart({ ...input, quantity: Number(input.quantity) || quantity });
      },

      removeItem: (id, variacao) =>
        set((state) => ({
          items: state.items.filter((i) => !(i.id === id && i.variacao === variacao)),
        })),

      updateQuantity: (id, quantity, variacao) =>
        set((state) => ({
          items:
            quantity <= 0
              ? state.items.filter((i) => !(i.id === id && i.variacao === variacao))
              : state.items.map((i) =>
                  i.id === id && i.variacao === variacao ? { ...i, quantity } : i
                ),
        })),

      clearCart: () => set({ items: [] }),

      totalItems: () => get().items.reduce((acc, i) => acc + i.quantity, 0),
    }),
    { name: 'bioflow_cart' }
  )
);
